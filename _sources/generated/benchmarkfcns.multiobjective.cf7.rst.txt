.. rubric:: Mathematical Definition

.. math::
   :nowrap:

   \begin{aligned}
f_1(\textbf{x}) &= x_1 + \frac{2}{|J_1|} \sum_{j \in J_1} [2y_j^2 - \cos(4\pi y_j) + 1] \\
f_2(\textbf{x}) &= (1 - x_1)^2 + \frac{2}{|J_2|} \sum_{j \in J_2} [2y_j^2 - \cos(4\pi y_j) + 1] \\
y_j &= \begin{cases} x_j - \cos(6\pi x_1 + \frac{j\pi}{n}) & j \in J_1 \\ x_j - \sin(6\pi x_1 + \frac{j\pi}{n}) & j \in J_2 \end{cases} \\
\text{Subject to: } & g_1, g_2 \ge 0 \quad \text{(Complex sinusoidal boundaries)} \\
& x_1 \in [0, 1], x_j \in [-2, 2]
\end{aligned}
