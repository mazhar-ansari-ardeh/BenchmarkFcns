   f_1 = x_1 + (\frac{1}{20} + 0.1) |\sin(20\pi x_1)| + \frac{2}{|J_1|} \sum_{j \in J_1} (2y_j^2 - \cos(4\pi y_j) + 1) \\
f_2 = 1 - x_1 + (\frac{1}{20} + 0.1) |\sin(20\pi x_1)| + \frac{2}{|J_2|} \sum_{j \in J_2} (2y_j^2 - \cos(4\pi y_j) + 1) \\
y_j = x_j - \sin(6\pi x_1 + \frac{j\pi}{n}) \\
J_1 = \{j | j \text{ is odd and } 2 \le j \le n\}, J_2 = \{j | j \text{ is even and } 2 \le j \le n\}
